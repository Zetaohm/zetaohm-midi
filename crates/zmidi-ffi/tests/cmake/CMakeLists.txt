# Checks the CMake package (DEC-019): on a host it builds and runs the C tests with CTest; for a
# bare-metal target it links tests/cmake/fw.c to check the staticlib and to print its size.
cmake_minimum_required(VERSION 3.22)
project(zmidi_cmake_check C)
# ZMIDI_CHECK_CI_ONLY: link fw_ci only, with the ci features of ZMIDI_CHECK_CI_FEATURES (for example
# the responder-only build ci-device;ci-property-exchange, REQ-FFI-CI-009).
option(ZMIDI_CHECK_CI_ONLY "Link fw_ci only, with ZMIDI_CHECK_CI_FEATURES" OFF)
set(ZMIDI_CHECK_CI_FEATURES "ci" CACHE STRING "The ci features of the fw_ci-only link")
if(ZMIDI_CHECK_CI_ONLY)
  set(ZMIDI_FEATURES ${ZMIDI_CHECK_CI_FEATURES})
else()
  set(ZMIDI_FEATURES io ump ble file ci net net-auth usb)
endif()
add_subdirectory(../.. zmidi)
if(CMAKE_CROSSCOMPILING)
  if(NOT ZMIDI_CHECK_CI_ONLY)
    add_executable(fw fw.c)
    target_compile_options(fw PRIVATE -Os -ffunction-sections -fdata-sections)
    target_link_options(fw PRIVATE -nostartfiles -e Reset_Handler -Wl,--gc-sections -Wl,-z,noexecstack)
    target_link_libraries(fw zmidi::zmidi)
    # OPT-038: a firmware with the fixed USB MIDI function (generated arrays, no run-time writer).
    add_executable(fw_usb_fixed fw_usb_fixed.c)
    target_include_directories(fw_usb_fixed PRIVATE ../c)
    target_compile_options(fw_usb_fixed PRIVATE -Os -ffunction-sections -fdata-sections)
    target_link_options(fw_usb_fixed PRIVATE -nostartfiles -e Reset_Handler -Wl,--gc-sections -Wl,-z,noexecstack)
    target_link_libraries(fw_usb_fixed zmidi::zmidi)
  endif()
  add_executable(fw_ci fw_ci.c)
  target_compile_options(fw_ci PRIVATE -Os -ffunction-sections -fdata-sections)
  target_link_options(fw_ci PRIVATE -nostartfiles -e Reset_Handler -Wl,--gc-sections -Wl,-z,noexecstack)
  target_link_libraries(fw_ci zmidi::zmidi)
else()
  enable_testing()
  add_executable(c_io ../c/io.c)
  target_compile_options(c_io PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_io zmidi::zmidi)
  add_test(NAME c_io COMMAND c_io)
  add_executable(c_ump ../c/ump.c)
  target_compile_options(c_ump PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_ump zmidi::zmidi)
  add_test(NAME c_ump COMMAND c_ump)
  add_executable(c_ble ../c/ble.c)
  target_compile_options(c_ble PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_ble zmidi::zmidi)
  add_test(NAME c_ble COMMAND c_ble)
  add_executable(c_file ../c/file.c)
  target_compile_options(c_file PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_file zmidi::zmidi)
  add_test(NAME c_file COMMAND c_file)
  add_executable(c_ci ../c/ci.c)
  target_compile_options(c_ci PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_ci zmidi::zmidi)
  add_test(NAME c_ci COMMAND c_ci)
  add_executable(c_pe ../c/pe.c)
  target_compile_options(c_pe PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_pe zmidi::zmidi)
  add_test(NAME c_pe COMMAND c_pe)
  add_executable(c_net ../c/net.c)
  target_compile_options(c_net PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_net zmidi::zmidi)
  add_test(NAME c_net COMMAND c_net)
  add_executable(c_usb ../c/usb.c)
  target_compile_options(c_usb PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_usb zmidi::zmidi)
  add_test(NAME c_usb COMMAND c_usb)
  add_executable(c_usb_desc ../c/usb_desc.c)
  target_compile_options(c_usb_desc PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_usb_desc zmidi::zmidi)
  add_test(NAME c_usb_desc COMMAND c_usb_desc)
  add_executable(c_usb_fixed ../c/usb_fixed.c)
  target_compile_options(c_usb_fixed PRIVATE -std=c99 -Wall -Wextra -Werror -pedantic)
  target_link_libraries(c_usb_fixed zmidi::zmidi)
  add_test(NAME c_usb_fixed COMMAND c_usb_fixed)
endif()
